\newcommand{\Field}[5]{%
  \coordinate (fieldbase) at (#1,#2);
  \filldraw[fill=white, draw=ink, line width=.55pt] (fieldbase) rectangle ++(#3,#4);
  \node[anchor=north west, font=\bfseries\fontsize{5.35}{6}\selectfont, text=ink]
    at ($(fieldbase)+(0.1mm,#4-0.1mm)$) {#5};
  \draw[line width=.25pt, color=muted] ($(fieldbase)+(1mm,1.8mm)$) -- ($(fieldbase)+(#3-1mm,1.8mm)$);
}

\newcommand{\Band}[2]{%
  \filldraw[fill=band, draw=ink, line width=.65pt] (4mm,#1) rectangle ++(\innerW,5mm);
  \node[anchor=west, font=\bfseries\fontsize{7.2}{8}\selectfont, text=ink] at (5mm,#1+2.5mm) {#2};
}

\newcommand{\Notes}[5]{%
  \draw[line width=.55pt, color=ink] (#1,#2) rectangle ++(#3,#4);
  \node[anchor=north west, font=\bfseries\fontsize{5.35}{6}\selectfont, text=muted]
    at ($(#1,#2)+(1mm,#4-1mm)$) {#5};
}
